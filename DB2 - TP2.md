## PARTIE 2


```cobol
WORKING-STORAGE SECTION.

     EXEC SQL  INCLUDE SQLCA    END-EXEC.
     EXEC SQL  INCLUDE TRACKS    END-EXEC.
     EXEC SQL  INCLUDE GENRES    END-EXEC.

     EXEC SQL
          DECLARE CGENRES CURSOR
          FOR
          SELECT G.NAME, T.NAME, T.UNITPRICE, T.MILLISECONDS
          FROM API4.GENRES G
          JOIN API4.TRACKS T ON G.GENREID = T.GENREID
          ORDER BY G.NAME, T.NAME
     END-EXEC.

77 ED-SQLCODE    PIC +9(9).
77 WS-ANO        PIC   9     VALUE ZERO.
01 WS-GENRE-ACTUEL PIC X(50) VALUE SPACES.
01 WS-COMPTEUR-GENRE PIC 9(4) VALUE 0.

01 WS-PISTE.
    05 FILLER PIC X(20) VALUE 'TITRE PISTE : '.
    05 ED-NOM-PISTE PIC X(50).
01 WS-TEMPS.
    05 FILLER PIC X(10) VALUE 'TEMPS : '.
    05 ED-MILLISECONDS PIC 9(9).
    05 FILLER PIC X(3) VALUE ' MS'.
01 WS-PRIX.
    05 FILLER PIC X(16) VALUE 'PRIX UNITAIRE : '.
    05 ED-UNITPRICE PIC ZZ9,99.
    05 FILLER PIC X(2) VALUE ' $'.
01 WS-GENRE.
    05 FILLER PIC X(8) VALUE 'GENRE : '.
    05 ED-GENRE PIC X(50).
01 WS-NB-PISTES.
    05 FILLER PIC X(9) VALUE 'NOMBRE : '.
    05 ED-NB-PISTES PIC 9(4).
01 WS-SEP PIC X(50) VALUE '=' ALL.

PROCEDURE DIVISION.
EXEC SQL
	OPEN CGENRES
END-EXEC
PERFORM TEST-SQLCODE

EXEC SQL
	FETCH CGENRES
	INTO : G-NAME, :T-NAME, :T-UNITPRICE, :T-MILLISECONDS
END-EXEC
PERFORM TEST-SQLCODE

PERFORM UNTIL SQLCODE NOT EQUAL ZERO
    IF G-NAME-TEXT(1:G-NAME-LEN) NOT EQUAL WS-GENRE-ACTUEL
        IF WS-GENRE-ACTUEL NOT EQUAL SPACES
            MOVE WS-COMPTEUR-GENRE TO ED-NB-PISTES
            DISPLAY WS-NB-PISTES
            DISPLAY WS-SEP
        END-IF
        MOVE G-NAME-TEXT(1:G-NAME-LEN) TO WS-GENRE-ACTUEL
        MOVE WS-GENRE-ACTUEL TO ED-GENRE
        DISPLAY WS-GENRE
        MOVE 0 TO WS-COMPTEUR-GENRE
    END-IF

    MOVE T-NAME-TEXT(1:T-NAME-LEN) TO ED-NOM-PISTE
    MOVE T-UNITPRICE TO ED-UNITPRICE
    MOVE T-MILLISECONDS TO ED-MILLISECONDS
    DISPLAY WS-PISTE
    DISPLAY WS-TEMPS
    DISPLAY WS-PRIX
    ADD 1 TO WS-COMPTEUR-GENRE

    EXEC SQL
	    FETCH CGENRES
	    INTO : G-NAME, :T-NAME, :T-UNITPRICE, :T-MILLISECONDS
    END-EXEC
    PERFORM TEST-SQLCODE
END-PERFORM

IF WS-GENRE-ACTUEL NOT EQUAL SPACES
    MOVE WS-COMPTEUR-GENRE TO ED-NB-PISTES
    DISPLAY WS-NB-PISTES
    DISPLAY WS-SEP
END-IF

EXEC SQL
CLOSE CGENRES
END-EXEC
PERFORM TEST-SQLCODE

GOBACK.

TEST-SQLCODE.
    MOVE SQLCODE TO ED-SQLCODE
    EVALUATE TRUE
      WHEN SQLCODE = ZERO OR SQLCODE = +100
       CONTINUE

      WHEN SQLCODE > ZERO AND SQLCODE NOT EQUAL +100
       DISPLAY 'WARNING - ' ED-SQLCODE

      WHEN OTHER
       DISPLAY 'ANOMALIE GRAVE - ' ED-SQLCODE
       PERFORM ABEND-PROG

    END-EVALUATE.
```

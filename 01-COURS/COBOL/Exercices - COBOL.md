77 WS-NB-EMP            PIC 9(5)        VALUE ZERO.
77 WS-TOTAL-SAL         PIC 9(9)V99     VALUE ZERO.
77 WS-MOY-SAL           PIC 9(7)V99     VALUE ZERO.

77 E-SALAIRE            PIC ZZ.ZZ9,99.
77 E-MOY                PIC ZZ.ZZ9,99.

    OPEN INPUT EMPLOYE
    IF WS-STATUS NOT = '00'
        DISPLAY 'ERREUR OUVERTURE FICHIER : ' WS-STATUS
        STOP RUN
    END-IF

    READ EMPLOYE
        AT END
            DISPLAY 'FICHIER EMPLOYES VIDE !'
            MOVE 1 TO WS-FIN
    END-READ

	PERFORM UNTIL WS-FIN = 1
        READ EMPLOYE
            AT END 
	            MOVE 1 TO WS-FIN
		    NOT AT END
				PERFORM LISTALLEMP
        END-READ
    END-PERFORM

    PERFORM CALCULSEMP

    CLOSE EMPLOYE

    STOP RUN.

 LISTALLEMP.
     ADD 1 TO WS-NB-EMP
     ADD SALAIRE-EMP TO WS-TOTAL-SAL
     MOVE SALAIRE-EMP TO E-SALAIRE
     DISPLAY NOM-EMP ' ' PREN-EMP ' ' E-SALAIRE.

 CALCULSEMP.
     DISPLAY 'NB EMPLOYES   : ' WS-NB-EMP
     IF WS-NB-EMP > 0
         COMPUTE WS-MOY-SAL = WS-TOTAL-SAL / WS-NB-EMP
         MOVE WS-MOY-SAL TO E-MOY
         DISPLAY 'SALAIRE MOYEN : ' E-MOY
     END-IF.

## TP3

```cobol
IDENTIFICATION DIVISION.
PROGRAM-ID. RECHKSDS.

ENVIRONMENT DIVISION.
CONFIGURATION SECTION.
SPECIAL-NAMES.
    DECIMAL-POINT IS COMMA.
INPUT-OUTPUT SECTION.
FILE-CONTROL.
    SELECT EMPLOYES ASSIGN TO EMPLOYES
        ORGANIZATION IS INDEXED
        ACCESS MODE IS RANDOM
        RECORD KEY IS CODE-EMP
        FILE STATUS IS WS-STATUS-EMP.

    SELECT FICHCLE ASSIGN TO FICHCLE
        ORGANIZATION IS SEQUENTIAL
        ACCESS MODE IS SEQUENTIAL
        FILE STATUS IS WS-STATUS-CLE.

DATA DIVISION.
FILE SECTION.

FD EMPLOYES.
01 ENR-EMPLOYE.
    05 CODE-EMP         PIC X(5).
    05 NOM-EMP          PIC X(15).
    05 PREN-EMP         PIC X(15).
    05 CP-EMP           PIC X(5).
    05 VILLE-EMP        PIC X(20).
    05 SALAIRE-EMP      PIC 9(5)V99.
    05 FILLER           PIC X(3).

FD FICHCLE.
01 ENR-CLE              PIC X(5).       

WORKING-STORAGE SECTION.
77 WS-STATUS-EMP        PIC X(2)        VALUE SPACES.
77 WS-STATUS-CLE        PIC X(2)        VALUE SPACES.
77 WS-FIN               PIC X           VALUE 'N'.

77 WS-NB-OK             PIC 9(5)        VALUE ZERO.
77 WS-NB-KO             PIC 9(5)        VALUE ZERO.
77 WS-NB-TOTAL          PIC 9(5)        VALUE ZERO.
77 WS-CUMUL-SAL         PIC 9(9)V99     VALUE ZERO.
77 WS-POURCENTAGE       PIC 9(3)V99     VALUE ZERO.

77 E-NB-OK              PIC Z(5).
77 E-NB-KO              PIC Z(5).
77 E-SALAIRE            PIC Z(5)9,99.
77 E-CUMUL              PIC Z(7)9,99.
77 E-POURCENT           PIC ZZ9,99.

PROCEDURE DIVISION.

    OPEN INPUT EMPLOYES
    IF WS-STATUS-EMP NOT = '00'
        DISPLAY 'ERREUR OUVERTURE EMPLOYES : ' WS-STATUS-EMP
        STOP RUN
    END-IF

    OPEN INPUT FICHCLE
    IF WS-STATUS-CLE NOT = '00'
        DISPLAY 'ERREUR OUVERTURE FICHCLE : ' WS-STATUS-CLE
        STOP RUN
    END-IF

    *> Premiere lecture
    READ FICHCLE
        AT END MOVE 'O' TO WS-FIN
    END-READ

    PERFORM UNTIL WS-FIN = 'O' OR ENR-CLE = '***'

        MOVE ENR-CLE TO CODE-EMP
        ADD 1 TO WS-NB-TOTAL

        READ EMPLOYES
            INVALID KEY
                DISPLAY 'EMPLOYE ' ENR-CLE ' INTROUVABLE'
                ADD 1 TO WS-NB-KO
            NOT INVALID KEY
                MOVE SALAIRE-EMP TO E-SALAIRE
                DISPLAY CODE-EMP ' ' NOM-EMP ' ' PREN-EMP
                        ' ' VILLE-EMP ' ' E-SALAIRE ' EUROS'
                ADD 1 TO WS-NB-OK
                ADD SALAIRE-EMP TO WS-CUMUL-SAL
        END-READ

        READ FICHCLE
            AT END MOVE 'O' TO WS-FIN
        END-READ

    END-PERFORM

    *> Calcul pourcentage
    IF WS-NB-TOTAL > 0
        COMPUTE WS-POURCENTAGE =
            (WS-NB-OK * 100) / WS-NB-TOTAL
    END-IF

    MOVE WS-NB-OK       TO E-NB-OK
    MOVE WS-NB-KO       TO E-NB-KO
    MOVE WS-CUMUL-SAL   TO E-CUMUL
    MOVE WS-POURCENTAGE TO E-POURCENT

    DISPLAY 'RECHERCHES BONNES      : ' E-NB-OK
    DISPLAY 'RECHERCHES INFRUCTUEUS : ' E-NB-KO
    DISPLAY 'CUMUL SALAIRES         : ' E-CUMUL ' EUROS'
    DISPLAY 'POURCENTAGE BONNES     : ' E-POURCENT ' %'

    CLOSE EMPLOYES
    CLOSE FICHCLE

    STOP RUN.
```

IDENTIFICATION DIVISION.

CONFIGURATION SECTION.

FILE-CONTROL.

        ACCESS MODE IS SEQUENTIAL
        RECORD KEY IS CODE-EMP
        FILE STATUS IS WS-STATUS.

DATA DIVISION.
FILE SECTION.

FD EMPLOYES.
01 ENR-EMPLOYE.
    05 CODE-EMP         PIC X(5).
    05 NOM-EMP          PIC X(15).
    05 PREN-EMP         PIC X(15).
    05 CP-EMP           PIC X(5).
    05 VILLE-EMP        PIC X(20).
    05 SALAIRE-EMP      PIC 9(5)V99.
    05 FILLER           PIC X(3).

WORKING-STORAGE SECTION.
77 WS-STATUS            PIC X(2)        VALUE SPACES.
77 WS-CLE               PIC X(5)        VALUE SPACES.
77 WS-FIN               PIC 9           VALUE 0.

77 WS-NB-EMP            PIC 9(5)        VALUE ZERO.
77 WS-CUMUL-SAL         PIC 9(9)V99     VALUE ZERO.
77 WS-MOY-SAL           PIC 9(7)V99     VALUE ZERO.

77 E-NB-EMP             PIC Z(5).
77 E-SALAIRE            PIC Z(5)9,99.
77 E-CUMUL              PIC Z(7)9,99.
77 E-MOY                PIC Z(5)9,99.

    OPEN INPUT EMPLOYES
    IF WS-STATUS NOT = '00'
        DISPLAY 'ERREUR OUVERTURE : ' WS-STATUS
        STOP RUN
    END-IF

    ACCEPT WS-CLE
    MOVE WS-CLE TO CODE-EMP

    START EMPLOYES KEY >= CODE-EMP
        INVALID KEY
            DISPLAY 'AUCUN ENREGISTREMENT A PARTIR DE ' WS-CLE
            CLOSE EMPLOYES
            STOP RUN
    END-START

    PERFORM UNTIL WS-FIN = 1
        READ EMPLOYES NEXT
            AT END
                MOVE 1 TO WS-FIN
            NOT AT END
                PERFORM TRAITEMENT-EMP
        END-READ
    END-PERFORM

    IF WS-NB-EMP > 0
        COMPUTE WS-MOY-SAL = WS-CUMUL-SAL / WS-NB-EMP
    END-IF

    MOVE WS-NB-EMP      TO E-NB-EMP
    MOVE WS-CUMUL-SAL   TO E-CUMUL
    MOVE WS-MOY-SAL     TO E-MOY

    DISPLAY '========================================'
    DISPLAY 'NOMBRE EMPLOYES  : ' E-NB-EMP
    DISPLAY 'CUMUL SALAIRES   : ' E-CUMUL  ' EUROS'
    DISPLAY 'SALAIRE MOYEN    : ' E-MOY    ' EUROS'
    DISPLAY '========================================'

    CLOSE EMPLOYES
    STOP RUN.

TRAITEMENT-EMP.
	ADD 1 TO WS-NB-EMP
	ADD SALAIRE-EMP TO WS-CUMUL-SAL
	MOVE SALAIRE-EMP TO E-SALAIRE
    DISPLAY CODE-EMP ' ' NOM-EMP ' ' PREN-EMP
             ' ' VILLE-EMP ' ' E-SALAIRE ' EUROS'.

## TP06

```cobol
IDENTIFICATION DIVISION.
PROGRAM-ID. A4TP06.

ENVIRONMENT DIVISION.
CONFIGURATION SECTION.
SPECIAL-NAMES.
    DECIMAL-POINT IS COMMA.
INPUT-OUTPUT SECTION.
FILE-CONTROL.
    SELECT EMPLOYE ASSIGN TO EMPLOYES
        ORGANIZATION IS SEQUENTIAL
        ACCESS MODE IS SEQUENTIAL
        FILE STATUS IS WS-STATUS-EMP.

    SELECT RECAP ASSIGN TO RECAP
        ORGANIZATION IS SEQUENTIAL
        ACCESS MODE IS SEQUENTIAL
        FILE STATUS IS WS-STATUS-RCP.

DATA DIVISION.
FILE SECTION.

FD EMPLOYE
    RECORDING MODE IS F
    RECORD CONTAINS 70 CHARACTERS.
01 ENR-EMPLOYE.
    05 CODE-EMP         PIC X(5).
    05 NOM-EMP          PIC X(15).
    05 PREN-EMP         PIC X(15).
    05 CP-EMP           PIC X(5).
    05 VILLE-EMP        PIC X(20).
    05 SALAIRE-EMP      PIC 9(5)V99.
    05 FILLER           PIC X(3).

FD RECAP
    BLOCK CONTAINS 0 RECORDS.
01 ENR-RECAP            PIC X(80).

WORKING-STORAGE SECTION.
01 WS-STATUS-EMP        PIC X(2).
01 WS-STATUS-RCP        PIC X(2).
01 WS-FIN               PIC 9           VALUE ZERO.
01 WS-EMP-PAGE          PIC 9           VALUE ZERO.

01 WS-LIGNE-ETOILES PIC X(51) VALUE ALL '*'.

01 WS-LIGNE-ENTETE PIC X(51) VALUE
       '* NOMS          * PRENOMS       *  SALAIRES    *'.

01 WS-LIGNE-EMP.
    05 FILLER           PIC X(2)        VALUE '* '.
    05 WS-EMP-NOM       PIC X(15).
    05 FILLER           PIC X(3)        VALUE ' * '.
    05 WS-EMP-PNM       PIC X(15).
    05 FILLER           PIC X(3)        VALUE ' * '.
    05 WS-EMP-SAL       PIC ZZ.ZZ9,99.
    05 FILLER           PIC X(4)        VALUE '   *'.

PROCEDURE DIVISION.

    OPEN INPUT EMPLOYE
    IF WS-STATUS-EMP NOT = '00'
        DISPLAY 'ERREUR OUVERTURE EMPLOYE : ' WS-STATUS-EMP
        STOP RUN
    END-IF

    OPEN OUTPUT RECAP
    IF WS-STATUS-RCP NOT = '00'
        DISPLAY 'ERREUR OUVERTURE RECAP : ' WS-STATUS-RCP
        STOP RUN
    END-IF

    PERFORM ENTETE

    READ EMPLOYE
        AT END
            DISPLAY 'FICHIER EMPLOYES VIDE !'
            MOVE 1 TO WS-FIN
    END-READ

    PERFORM UNTIL WS-FIN = 1
        READ EMPLOYE
            AT END
                MOVE 1 TO WS-FIN
            NOT AT END
                PERFORM LIGNE-EMP
        END-READ
    END-PERFORM

    CLOSE EMPLOYE
    CLOSE RECAP
    STOP RUN.

 ENTETE.
    WRITE ENR-RECAP FROM WS-LIGNE-ETOILES
    WRITE ENR-RECAP FROM WS-LIGNE-ENTETE
	WRITE ENR-RECAP FROM WS-LIGNE-ETOILES
    MOVE ZERO TO WS-EMP-PAGE.

 LIGNE-EMP.
    IF WS-EMP-PAGE = 5
	    MOVE WS-LIGNE-ETOILES TO ENR-RECAP
        WRITE ENR-RECAP BEFORE ADVANCING PAGE
        PERFORM ENTETE
    END-IF

    IF WS-EMP-PAGE <> 0
	    MOVE WS-LIGNE-ETOILES TO ENR-RECAP
        WRITE ENR-RECAP
    END-IF

    MOVE NOM-EMP       TO WS-EMP-NOM
    MOVE PREN-EMP      TO WS-EMP-PNM
    MOVE SALAIRE-EMP   TO WS-EMP-SAL
    MOVE WS-LIGNE-EMP  TO ENR-RECAP
    WRITE ENR-RECAP

    ADD 1 TO WS-EMP-PAGE.
```
